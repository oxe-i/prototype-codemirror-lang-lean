# A character literal holds one character, so a stray quote is an operator

xs[i]'h

==>

Program(Identifier, Brackets(Identifier), Operator, Identifier)

# A character literal may follow an operator without a space

xs.filter (·!='"') |> f (c=='a') (d≠'\n')

==>

Program(
  Identifier,
  Parens(Operator, Char),
  Operator,
  Identifier,
  Parens(Identifier, Operator, Char),
  Parens(Identifier, Operator, CharEscape))
